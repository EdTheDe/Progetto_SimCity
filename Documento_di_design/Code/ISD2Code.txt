sequenceDiagram
    autonumber
    participant Thread as SimulationThread
    participant Engine as SimulationEngine
    participant Grid as UrbanGrid
    participant Entity as UrbanEntity
    participant Stato as StatoCitta
    participant Politica as PoliticaStrategy
    participant Event as RandomEvent
    participant Obs as CityObserver (TopBar, etc.)

    Thread->>Engine: run() loop
    
    loop Ogni speedDelay millisecondi
        Engine->>Engine: applicaTick()
        activate Engine
        
        Engine->>Grid: Itera su tutte le Celle
        activate Grid
        Grid->>Entity: processTick()
        Entity-->>Grid: calcolaTasse / Consumi
        Grid-->>Engine: restituisce TickStats
        deactivate Grid
        
        Engine->>Stato: aggiorna(TickStats)
        
        opt Se c'è una Politica attiva
            Engine->>Politica: applica(stato, griglia)
            Politica->>Stato: applicaBonusMalus()
        end
        
        opt Se scatta un evento casuale (bassa % o trigger)
            Engine->>Event: trigger(griglia, stato)
            Event->>Stato: infliggeDanni()
            Engine->>Obs: onEventTriggered(titolo, desc)
        end
        
        Engine->>Engine: calcolaParametriGlobali() (felicità, lavoro, ecc.)
        
        Engine->>Engine: notifyObservers()
        Engine->>Obs: onCityUpdate(stato, griglia)
        
        deactivate Engine
    end
