sequenceDiagram
    autonumber
    actor Utente as Utente
    participant Sistema as Sistema (SimCity)

    %% --- GESTIONE EDIFICI E POLITICHE ---
    Note over Utente,Sistema: Fase 1: Gestione della Città
    
    Utente->>Sistema: selezionaEdificioDaCostruire(tipo)
    Sistema-->>Utente: memorizzaSelezioneEMostraFeedback(cursore)
    
    Utente->>Sistema: cliccaCellaMappa(x, y)
    alt Costruzione non valida (spazio occupato, fondi, ecc.)
        Sistema-->>Utente: lanciaEccezioneEImpedisceAzione()
    else Costruzione valida
        Sistema-->>Utente: deduciFondiAggiungiEdificioEAggiornaMappa()
    end

    Utente->>Sistema: clickSecondarioCella(x, y)
    Sistema-->>Utente: mostraMenuContestuale(Migliora, Demolisci)
    
    alt Azione: Miglioramento
        Utente->>Sistema: selezionaMigliora()
        alt Requisiti soddisfatti
            Sistema-->>Utente: deduceFondiEAggiornaLivelloETexture()
        else Requisiti mancanti
            Sistema-->>Utente: lanciaEccezioneEImpedisceAzione()
        end
    else Azione: Demolizione
        Utente->>Sistema: selezionaDemolisci()
        Sistema-->>Utente: rimuoviEdificioLiberaCellaEAggiornaMappa()
    end

    Utente->>Sistema: espandiMenuPolitiche()
    Sistema-->>Utente: mostraListaPolitiche()
    Utente->>Sistema: impostaPoliticaCittadina(nuovaPolitica)
    Sistema-->>Utente: ricalcolaModificatoriEAggiornaUI()

    %% --- SIMULAZIONE E TEMPO ---
    Note over Utente,Sistema: Fase 2: Controllo del Tempo e Simulazione
    
    Utente->>Sistema: interagisciConPannelloTempo(velocita/play/pausa)
    Sistema-->>Utente: alteraFramerateTimeline()
    
    loop Ciclo di Simulazione Continuo (Tick)
        Sistema-->>Utente: calcolaDeltaEAggiornaStatisticheGlobali()
        Sistema-->>Utente: aggiornaVisualizzazioneStatoEdifici(segnalaInattivita)
        
        opt Rilevato Evento Casuale (Disastro/Bonus)
            Sistema-->>Utente: mettiInPausaEMostraPopupEvento(dettagli)
            Utente->>Sistema: acknowledgeEvento(conferma)
            Sistema-->>Utente: applicaMalusERiprendiSimulazione()
        end
    end

    %% --- PERSISTENZA ---
    Note over Utente,Sistema: Fase 3: Persistenza (Menu Impostazioni)
    
    Utente->>Sistema: apriMenuImpostazioni()
    Sistema-->>Utente: mostraPopupImpostazioni()
    
    alt Salvataggio Partita
        Utente->>Sistema: cliccaEsportaSalvataggio()
        Sistema-->>Utente: apreDialogSceltaDestinazione()
        Utente->>Sistema: confermaPercorsoENomeFile()
        Sistema-->>Utente: serializzaStatoENotificaCompletamento()
    else Caricamento Partita
        Utente->>Sistema: cliccaImportaCaricamento()
        Sistema-->>Utente: metteInPausaEApreDialogSceltaFile()
        Utente->>Sistema: selezionaFileJSON()
        Sistema-->>Utente: pulisceGrigliaCaricaDatiERigeneraMappa()
    end
