sequenceDiagram
    autonumber
    actor Utente as Utente
    participant Sistema as Sistema (SimCity)

    %% --- GESTIONE EDIFICI E POLITICHE ---
    Note over Utente,Sistema: Fase 1: Gestione della Città
    
    Utente->>Sistema: selezionaEdificioDaCostruire(tipo)
    Utente->>Sistema: posizionaEdificio(x, y)
    alt Costruzione non valida (es. fondi insufficienti, spazio occupato)
        Sistema-->>Utente: mostraErrore(motivo)
    else Costruzione valida
        Sistema-->>Utente: aggiornaMappaEFinanze()
    end

    Utente->>Sistema: miglioraEdificio(x, y)
    alt Requisiti di miglioramento soddisfatti
        Sistema-->>Utente: aggiornaLivelloETexture()
    else
        Sistema-->>Utente: mostraErrore(motivo)
    end
    
    Utente->>Sistema: demolisciEdificio(x, y)
    Sistema-->>Utente: rimuoviEdificioDallaMappa()

    Utente->>Sistema: impostaPoliticaCittadina(nuovaPolitica)
    Sistema-->>Utente: confermaCambioPolitica()

    %% --- SIMULAZIONE E TEMPO ---
    Note over Utente,Sistema: Fase 2: Controllo del Tempo e Simulazione
    
    Utente->>Sistema: impostaVelocitaTempo(moltiplicatore)
    Utente->>Sistema: avviaSimulazione()
    
    loop Ciclo di Simulazione Continuo (Tick)
        Sistema-->>Utente: aggiornaStatisticheGlobali(finanze, popolazione, felicità)
        
        opt Evento Casuale o Rischio Game Over
            Sistema-->>Utente: mostraNotificaImportante(messaggio)
        end
    end
    
    Utente->>Sistema: mettiInPausa()
    Sistema-->>Utente: sospendeAvanzamentoTempo()

    %% --- PERSISTENZA ---
    Note over Utente,Sistema: Fase 3: Persistenza (Menu)
    
    alt Salvataggio Partita
        Utente->>Sistema: richiedeSalvataggio(nomePartita)
        Sistema-->>Utente: confermaSalvataggioRiuscito()
    else Caricamento Partita
        Utente->>Sistema: richiedeCaricamento(filePartita)
        Sistema-->>Utente: ripristinaMappaEStatistiche()
    end
